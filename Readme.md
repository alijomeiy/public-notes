# Compression

```bash
#!/usr/bin/env bash 
set -o pipefail

IMAGE="myimage:latest" 
OUTPUT="myimage.tar" 

if command -v xz >/dev/null 2>&1; then
	echo "Using xz..."
	docker image save "$IMAGE" | xz -T0 -9 -c > "${OUTPUT}.xz" 
	# Extraction: xz -dc myimage.tar.xz | docker load


elif command -v zstd >/dev/null 2>&1; then
	echo "Using zstd..." 
	docker image save "$IMAGE" | zstd -T0 -19 -o "${OUTPUT}.zst" 
	# Extraction: zstd -dc myimage.tar.zst | docker load 
	
elif command -v gzip >/dev/null 2>&1; then
	echo "Using gzip..." docker image save "$IMAGE" | gzip -9 > "${OUTPUT}.gz" 
	# Extraction: gzip -dc myimage.tar.gz | docker load
```

**توجه‌:** به ترتیب براساس مقدار فشرده‌سازی با سرعت بالا مرتب‌شده اما ممکن است بالاتری‌ها نصب نباشد و نیاز به نصب بسته باشند
